عينة معمارية: معالجة الطلبات اللحظية وتوزيع الحمل

نموذج تطبيقي لكيفية كتابة كود نظيف (Clean Code) في NestJS / TypeScript مع آلية الـ Caching بواسطة Redis والتحقق المسبق من الصلاحيات.

src/modules/orders/orders.controller.ts
TypeScript Production Ready 
import { Controller, Post, Body, UseGuards, UseInterceptors } from '@nestjs/common';
import { AuthGuard } from '@app/auth/guards';
import { RateLimitInterceptor } from '@app/common/interceptors';
import { OrdersService } from './orders.service';
import { CreateOrderDto } from './dto/create-order.dto';

@Controller('api/v2/orders')
@UseGuards(AuthGuard)
export class OrdersController {
  constructor(private readonly ordersService: OrdersService) {}

  @Post('process-checkout')
  @UseInterceptors(new RateLimitInterceptor({ max: 100, windowMs: 60000 }))
  async checkout(@Body() dto: CreateOrderDto) {
    // 1. Idempotency validation & Redis atomic lock
    await this.ordersService.acquireDistributedLock(dto.idempotencyKey);
    
    // 2. Execute transactional persistence & broadcast event to Kafka
    const transactionResult = await this.ordersService.executeTransaction(dto);
    
    return {
      statusCode: 201,
      orderId: transactionResult.id,
      processingTimeMs: transactionResult.latency,
      status: 'ACCEPTED_IN_QUEUE'
    };
  }
}
